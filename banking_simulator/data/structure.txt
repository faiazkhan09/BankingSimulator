Future<List<Map<String, dynamic>>> readAccountInfo() async {
  final accountInfo = File('banking_simulator/data/accounts.txt');

  try {
    // Check if file exists
    if (await accountInfo.exists()) {
      // Read entire file as String
      String accountDetails = await accountInfo.readAsString();

      // If file is empty, return empty list
      if (accountDetails.trim().isEmpty) {
        return [];
      }

      // Decode the JSON list
      List<dynamic> decodedData = jsonDecode(accountDetails);

      // Convert dynamic list into List<Map<String, dynamic>>
      return decodedData.cast<Map<String, dynamic>>();
    } else {
      return [];
    }
  } catch (e) {
    print('No file found or invalid JSON.');
    return [];
  }
}
